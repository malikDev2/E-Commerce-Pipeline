Five stages, once a day

Every run walks the same path — the sequence rarely changes, only the volume does.

Extract
Pull raw sales, order, and product data from marketplace APIs and the order DB. Land untouched files in S3, partitioned by ingest date.

Load
Write conformed facts and dimensions into Snowflake. Cluster fact_orders on order_date for fast date-range scans.

Report

dbt-style SQL

Tech stack

Chosen for one job each — orchestration, storage, compute, and reporting — with clean handoffs between them.

Orchestration

Prefect

Schedules the daily flow, retries failed tasks, and pages the on-call channel when a run breaks its SLA.

Chosen over a bare cron job for built-in retry logic and a run history you can actually query.
Compute & packaging

Docker

Every task — extract, stage, transform, load — runs from the same container image, so local dev matches production.

Removes "works on my machine" drift between the laptop and the nightly run.
Landing storage

AWS S3

Holds every raw file exactly as it arrived, partitioned by source and ingest date, before anything is parsed or typed.

Cheap, durable, and replayable — a bad transform never means a lost source file.
Staging database

PostgreSQL

A typed, deduplicated holding area where raw rows are validated before they're trusted enough to transform.

Gives the pipeline a place to enforce constraints before data reaches the warehouse.
Transform language

Python

Handles currency normalization, return netting, and slowly-changing-dimension logic that's awkward in pure SQL.

One language across extract, transform, and orchestration — fewer context switches.
Warehouse

Snowflake

Hosts the conformed star schema — fact_orders, fact_returns, and the shared dimensions — that finance reports run against.

Separates storage from compute, so a heavy finance report never slows down the nightly load.

"""Pull raw order and product data and land it in S3, untouched."""
import boto3
import requests
from datetime import date

RAW_BUCKET = "ledgerline-raw"

def fetch_orders(marketplace_api: str, api_key: str) -> bytes:
    resp = requests.get(
        marketplace_api,
        headers={"Authorization": f"Bearer {api_key}"},
        params={"since": date.today().isoformat()},
        timeout=30,
    )
    resp.raise_for_status()
    return resp.content

def land_in_s3(payload: bytes, source: str) -> str:
    key = f"raw/{source}/{date.today().isoformat()}/orders.json"
    boto3.client("s3").put_object(Bucket=RAW_BUCKET, Key=key, Body=payload)
    return key

def run(sources: list[dict]) -> list[str]:
    landed = []
    for src in sources:
        payload = fetch_orders(src["api"], src["api_key"])
        landed.append(land_in_s3(payload, src["name"]))
    return landed

"""Load raw S3 files into typed, deduplicated PostgreSQL staging tables."""
import hashlib
import json
import psycopg2

INSERT_SQL = """
    INSERT INTO staging.orders_raw (order_id, payload, payload_hash, ingested_at)
    VALUES (%s, %s, %s, now())
    ON CONFLICT (order_id) DO NOTHING
"""

def hash_payload(payload: dict) -> str:
    blob = json.dumps(payload, sort_keys=True).encode()
    return hashlib.sha256(blob).hexdigest()

def stage_file(conn, records: list[dict]) -> int:
    inserted = 0
    with conn.cursor() as cur:
        for record in records:
            cur.execute(
                INSERT_SQL,
                (record["order_id"], json.dumps(record), hash_payload(record)),
            )
            inserted += cur.rowcount
    conn.commit()
    return inserted

"""Reshape staged rows into the conformed fact_orders star schema."""
import pandas as pd

FX_RATES = {"USD": 1.0, "EUR": 1.09, "GBP": 1.27}

def normalize_currency(df: pd.DataFrame) -> pd.DataFrame:
    df["net_revenue_usd"] = df["net_revenue"] * df["currency"].map(FX_RATES)
    return df

def net_returns(orders: pd.DataFrame, returns: pd.DataFrame) -> pd.DataFrame:
    refunds = returns.groupby("order_id")["refund_amount"].sum()
    orders["net_revenue"] = orders["gross_revenue"] - orders["order_id"].map(refunds).fillna(0)
    return orders

def build_fact_orders(orders: pd.DataFrame, returns: pd.DataFrame) -> pd.DataFrame:
    orders = net_returns(orders, returns)
    orders = normalize_currency(orders)
    return orders[[
        "order_id", "order_date", "customer_id", "product_id",
        "quantity", "net_revenue_usd", "status",
    ]]

"""Write conformed dimensions and facts into the Snowflake warehouse."""
import snowflake.connector

def load_fact_orders(cursor, stage_path: str) -> None:
    cursor.execute(f"""
        COPY INTO warehouse.fact_orders
        FROM '{stage_path}'
        FILE_FORMAT = (TYPE = PARQUET)
        MATCH_BY_COLUMN_NAME = CASE_INSENSITIVE
    """)

def refresh_clustering(cursor) -> None:
    cursor.execute("ALTER TABLE warehouse.fact_orders CLUSTER BY (order_date)")

def run(stage_path: str, account: str, warehouse: str) -> None:
    conn = snowflake.connector.connect(account=account, warehouse=warehouse)
    with conn.cursor() as cur:
        load_fact_orders(cur, stage_path)
        refresh_clustering(cur)
    conn.close()
